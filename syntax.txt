// 2. STRUCTS (Declaration + Implementation)

type Point {
    x: i32;
    y: i32;
}

impl Point {
    // Static constructor
    pub static func init(x: i32, y: i32) -> Point {
        return Point { x, y };
    }

    // Static factory
    pub static func origin() -> Point {
        return Point { 0, 0 };
    }

    // Immutable-reference method
    pub func magnitude(&self) -> f64 {
        return math::sqrt((self.x * self.x + self.y * self.y));
    }

    // Mutable-reference method
    pub func translate(&mut self, dx: i32, dy: i32) {
        self.x += dx;
        self.y += dy;
    }

    // By-value method
    pub func get_x(self) -> i32 {
        return self.x;
    }

    // Method with additional reference parameter
    pub func transpose(&mut self, other: &Point) {
        self.x += other.x;
        self.y += other.y;
    }
}

// ============================================================
// 3. STRUCT WITH STATIC VARIABLES
// ============================================================

type Counter {
    count: i32;
}

impl Counter {
    pub static let MAX: i32 = 100;
    pub static let mut total: i32 = 0;

    pub static func init() -> Counter {
        return Counter { 0 };
    }

    pub func increment(&mut self) {
        self.count += 1;
        Counter::total += 1;
    }
}

// Simple function with no parameters
func no_params() -> void {
    io::stdout().write("Hello\n");

// Function with parameters
func add(a: i32, b: i32) -> i32 {
    return a + b;
}

func add(a: f64, b: f64) -> f64 {
    return a + b;
}

// Function with mutable parameters
func add_mut(mut a: i32, mut b: i32) -> i32 {
    a += 2;
    return a + b;
}

// Function with reference parameters

    value = value * 2;
}

// Function with immutable reference parameters

func expressions() -> i32 {
    // Arithmetic
    let a = 1 + 2;
    let b = 10 - 3;
    let c = 4 * 5;
    let d = 20 / 4;
    let e = 10 % 3;

    // Compound assignment
    let mut f = 10;
    f += 1;
    f -= 1;
    f *= 2;
    f /= 2;
    f %= 3;

    // Bitwise
    let g = 0b1010 & 0b1100;
    let h = 0b1010 | 0b1100;
    let i = 0b1010 ^ 0b1100;
    let j = 1 << 4;
    let k = 16 >> 2;

    // Unary
    let l = -42;
    let m = !true;
    let n = ~0xFF;

    // Comparison
    let o = 1 == 2;
    let p = 1 != 2;
    let q = 1 < 2;
    let r = 1 <= 2;
    let s = 1 > 2;
    let t = 1 >= 2;

    // Logical
    let u = true && false;
    let v = true || false;

    // Ternary
    let w = true ? 1 : 2;

    // Indexing
    let arr: [i32; 3] = [1, 2, 3];
    let x = arr[0];
    arr[0] = 42;      // sema: arr is not mutable

    // Member access
    let point = Point { x: 3, y: 4 };
    let px = point.x;

    // Function calls
    let y = add(1, 2);
    let z = math::sqrt(4.0);

    // Method calls
    let mag = point.magnitude();

    // Static method calls
    let origin = Point::origin();

    // Literals
    let int_lit = 42;
    let hex_lit = 0xFF;
    let bin_lit = 0b1010;
    let oct_lit = 0o777;
    let float_lit = 3.14;
    let sci_lit = 1.0e9;
    let suffix_lit = 42i64;

    // Pointer comparison
    let pa: *i32 = nullptr;
    let is_null = pa == nullptr;

    return 0;
}
