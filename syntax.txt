type Point {
    x: i32;
    y: i32;
}

impl Point {
    // Static constructor
    pub static func Point(x: i32, y: i32) -> Point {
        return Point { x, y };
    }

    // Static factory
    pub static func origin() -> Point {
        return Point { 0, 0 };
    }

    // Immutable method
    pub func magnitude(self: &Point) -> f64 {
        return math::sqrt((self.x * self.x + self.y * self.y));
    }

    // Mutable method
    pub func translate(self: &mut Point, dx: i32, dy: i32) {
        self.x += dx;
        self.y += dy;
    }

    // Operator overload: binary +
    pub operator+(self: &Point, other: &Point) -> Point {
        return Point { x: self.x + other.x, y: self.y + other.y };
    }

    // Operator overload: binary -
    pub operator-(self: &Point, other: &Point) -> Point {
        return Point { x: self.x - other.x, y: self.y - other.y };
    }

    // Operator overload: unary -
    pub operator-(self: &Point) -> Point {
        return Point { x: -self.x, y: -self.y };
    }

    // Operator overload: +=
    pub operator+=(self: &mut Point, other: &Point) -> &mut Point {
        self.x += other.x;
        self.y += other.y;
        return self;
    }

    // Operator overload: ==
    pub operator==(self: &Point, other: &Point) -> bool {
        return self.x == other.x && self.y == other.y;
    }

    // Operator overload: !=
    pub operator!=(self: &Point, other: &Point) -> bool {
        return !(self == other);
    }

    // Destructor
    pub func drop(self: &Point) {
        // cleanup
    }
}

type Counter {
    count: i32;
}

impl Counter {
    pub static let MAX: i32 = 100;
    pub static let mut total: i32 = 0;

    pub static func init() -> Counter {
        return Counter { 0 };
    }

    pub func increment(self: &mut Counter) {
        self.count += 1;
        Counter::total += 1;
    }
}
