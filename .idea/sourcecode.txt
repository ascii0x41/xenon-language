// ==== Level 1. Fibonacci Sequence ====
func main() -> i32 {
    var nums: [i32] = [0, 1];  // Explicit type

    foreach i: 32 in step(1, 101) {
        nums.append(nums[i] + nums[i - 1]);
    }

    io::writeln($"Fibonacci Sequence ({nums.Size} terms):")
    io::writeln(nums.join(", "));
    return 0;
}


// ==== Level 2. Rocket Simulator ====
enum LaunchError { Success, FuelLow, EngineFailure }

class Rocket<T> {
    let name: str
    var fuel_cap: f64;
    var is_launched: bool;  // Default false
    pub var payload: T;

    pub func Rocket(str name, double fuel, T payload) {
        this.name = name;
        this.fuel = fuel;
        this.payload = payload;
    }

    pub mut func launch() -> void!LaunchError {
        Hardware::Ignite()?; // Early return
        fuel -= 75.0
        is_launched = true
    }
}

class Satellite {
    let name: str;
    
    pub func Satellite(n: str) {
        this.name = n;
    }

    pub func activate() {
        io::writeln($"Hello from ${this.name}!")
    }
}

func main() -> i32 {
    var rocket: Rocket<Satellite> = Rocket("MilkyWay1", 100.0, Satellite("Coffee"));

    let result = rocket.launch();
    
    if result.is_err() {
        io::writeln($"Launch failed: ${result.err()}")
        return 1;
    }

    rocket.payload.activate()
    return 0;
}


// ==== Level 3. Kinematics ====

class Vector3 {
    pub var x: f64;
    pub var y: f64;
    pub var z: f64;

    pub func Vector3(x_: f64, y_: f64, z_: f64) {
        this.x = x_;
        this.y = y_;
        this.z = z_;
    }

    pub func Vector3() {
        return Vector3(0.0, 0.0, 0.0);
    }

    pub func cross(Vector3 other) -> Vector3 {
        return Vector3(
            this.y * other.z - this.z * other.y,
            this.z * other.x - this.x * other.z,
            this.x * other.y - this.y * other.x
        )
    }

    operator+(Vector3 other) -> Vector3 {
        return Vector3(
            this.x + other.x,
            this.y + other.y,
            this.z + other.z
        );
    }

    operator*(double scalar) -> Vector3 {
        return Vector3(
            this.x * scalar,
            this.y * scalar,
            this.z * scalar
        );
    }

    operator/(scalar: f64) {
        return Vector3(
            this.x / scalar,
            this.y / scalar,
            this.z / scalar
        );
    }

    operator-() -> Vector3 {
        return Vector3(-this.x, -this.y, -this.z);
    }
}

impl ToString for Vector3 {
    pub func to_string() -> str {
        return $"x: {this.x}, y: {this.y}, z: {this.z}";
    }
}


class RigidBody {
    var position: Vector3;
    var velocity: Vector3;
    var acceleration: Vector3;

    var mass: f64;

    pub func RigidBody(x_0_: ref Vector3, m_: f64) {
        position = x_0_;
        mass = m_;
    }
    
    pub mut func update(dt: f64) -> void {
        assert(dt > 0);
        velocity += acceleration * dt;
        position += velocity * dt;
        acceleration = Vector3(); // Reset
    }
    
    pub mut func apply_force(force: ref Vector3) -> void {
        acceleration += force / mass;
    }
}

func main() -> i32 {
    var body = RigidBody(
        Vector3(0, 10, 0),
        2.0
    );

    let gravity = Vector3(0, -9.8, 0);
    let timestep = 0.016f32;  // ~60fps

    foreach frame: i32 in step(0, 100) {
        body.apply_force(gravity * body.mass)
        body.update(timestep)
        io::writeln($"Frame {frame}: Position = {body.position}")
    }
    return 0;
}
